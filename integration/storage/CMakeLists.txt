cmake_minimum_required(VERSION 3.25)

project(quant_storage_slice LANGUAGES CXX)

find_package(Arrow REQUIRED CONFIG)
find_package(Parquet REQUIRED CONFIG)

if(TARGET Arrow::arrow_shared)
  set(QUANT_ARROW_TARGET Arrow::arrow_shared)
elseif(TARGET Arrow::arrow_static)
  set(QUANT_ARROW_TARGET Arrow::arrow_static)
elseif(TARGET arrow_shared)
  set(QUANT_ARROW_TARGET arrow_shared)
elseif(TARGET arrow_static)
  set(QUANT_ARROW_TARGET arrow_static)
else()
  message(FATAL_ERROR "Apache Arrow CMake package did not provide a supported Arrow target")
endif()

if(TARGET Parquet::parquet_shared)
  set(QUANT_PARQUET_TARGET Parquet::parquet_shared)
elseif(TARGET Parquet::parquet_static)
  set(QUANT_PARQUET_TARGET Parquet::parquet_static)
elseif(TARGET parquet_shared)
  set(QUANT_PARQUET_TARGET parquet_shared)
elseif(TARGET parquet_static)
  set(QUANT_PARQUET_TARGET parquet_static)
else()
  message(FATAL_ERROR "Apache Parquet CMake package did not provide a supported Parquet target")
endif()

enable_testing()

add_executable(quant_storage_tests storage_tests.cpp)
target_compile_features(quant_storage_tests PRIVATE cxx_std_20)
target_compile_options(quant_storage_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(quant_storage_tests PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
target_link_libraries(quant_storage_tests PRIVATE "${QUANT_ARROW_TARGET}" "${QUANT_PARQUET_TARGET}")
add_test(NAME quant_storage_tests COMMAND quant_storage_tests)

add_executable(quant_storage_replay_boundary_tests replay_boundary_tests.cpp)
target_compile_features(quant_storage_replay_boundary_tests PRIVATE cxx_std_20)
target_compile_options(quant_storage_replay_boundary_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(quant_storage_replay_boundary_tests PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
target_link_libraries(quant_storage_replay_boundary_tests PRIVATE "${QUANT_ARROW_TARGET}")
add_test(NAME quant_storage_replay_boundary_tests COMMAND quant_storage_replay_boundary_tests)
